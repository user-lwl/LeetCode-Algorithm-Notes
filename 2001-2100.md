```java
class Solution {
    public int[] findOriginalArray(int[] changed) {
        if (changed.length % 2 == 1) {
            return new int[]{};
        }
        int[] result = new int[changed.length / 2];
        Map<Integer, Integer> tempMap = new HashMap<>();
        int index = 0;
        Arrays.sort(changed);
        for (int i = changed.length - 1; i >= 0; i--) {
            int key = changed[i];
            int twoKey = 2 * key;
            if (tempMap.containsKey(twoKey)) {
                result[index++] = key;
                int twoVal = tempMap.get(twoKey);
                if (twoVal == 1) {
                    tempMap.remove(twoKey);
                } else {
                    tempMap.put(twoKey, twoVal - 1);
                }
            } else {
                tempMap.put(key, tempMap.getOrDefault(key, 0) + 1);
            }
        }
        return tempMap.size() == 0 ? result : new int[]{};
    }
}
```

```java
    }
    
    public void add(int[] point) {
        int heng = point[1];
        List<int[]> tempList = hengMap.getOrDefault(heng, null);
        if (tempList == null) {
            tempList = new ArrayList<>();
            hengMap.put(heng, tempList);
        }
        tempList.add(point);

        int heng = point[1];
        List<int[]> tempList = hengMap.getOrDefault(heng, new ArrayList<>());
        int result = 0;
        for (int i = 0; i < tempList.size(); i++) {
            int[] tempInt = tempList.get(i);
            int len = Math.abs(point[0] - tempInt[0]);
            if (len == 0) {
                continue;
            }
            int[] dot1_1 = new int[]{tempInt[0], tempInt[1] + len};
            int[] dot1_2 = new int[]{point[0], point[1] + len};
            int[] dot2_1 = new int[]{tempInt[0], tempInt[1] - len};
            int[] dot2_2 = new int[]{point[0], point[1] - len};

            String dot1_2Str = Arrays.toString(dot1_2);
            String dot2_1Str = Arrays.toString(dot2_1);
            String dot2_2Str = Arrays.toString(dot2_2);

            int dot2_1Cnt = indexMap.getOrDefault(dot2_1Str, 0);
            int dot2_2Cnt = indexMap.getOrDefault(dot2_2Str, 0);
            result += dot1_1Cnt * dot1_2Cnt + dot2_1Cnt * dot2_2Cnt;
        }
        return result;
    }
```

# 2033.获取单值网格的最小操作数

```java
class Solution {
    public int minOperations(int[][] grid, int x) {
        int[] tempArray = new int[grid.length * grid[0].length];
        int index = 0;
        for (int i = 0; i < grid.length; i++) {
            for (int j = 0; j < grid[i].length; j++) {
                tempArray[index++] = grid[i][j];
            }
        }
        Arrays.sort(tempArray);
        long[] prefix = new long[tempArray.length + 1];
        for (int i = 1; i < prefix.length; i++) {
            prefix[i] = prefix[i - 1] + tempArray[i - 1];
            if (i > 1 && (tempArray[i - 1] - tempArray[i - 2]) % x != 0) {
                return -1;
            }
        }
        long total = prefix[prefix.length - 1];
        long result = Integer.MAX_VALUE;
        for (int i = 1; i < prefix.length; i++) {
            long thisNumber = tempArray[i - 1];
            long left = prefix[i - 1];
            long right = total - prefix[i];
            long leftNumber = i - 1;
            long rightNumber = prefix.length - i - 1;
            long leftCnt = (leftNumber * thisNumber - left) / x;
            long rightCnt = (right - rightNumber * thisNumber) / x;
            long tempRes = leftCnt + rightCnt;
            result = Math.min(result, tempRes);
        }
        return (int) result;
    }
}
```

# 2037.使每位学生都有座位的最少移动次数

```java
class Solution {
    public int minMovesToSeat(int[] seats, int[] students) {
        heapSort(seats);
        heapSort(students);
        int len = seats.length;
        int sum = 0;
        for (int i = 0; i < len; i++) {
            sum += Math.abs(students[i] - seats[i]);
        }
        return sum;
    }
    public void swap(int[] nums, int left, int right) {
        int temp = nums[left];
        nums[left] = nums[right];
        nums[right] = temp;
    }
    public void maxHeapify(int[] nums, int left, int right) {
        int son = left * 2;
        while (son <= right) {
            if (son + 1 <= right && nums[son + 1] > nums[son]) {
                son++;
            }
            if (nums[son] > nums[left]) {
                swap(nums, son, left);
                left = son;
                son = son * 2;
            } else {
                break;
            }
        }
    }
    public void heapSort(int[] nums) {
        int len = nums.length - 1;
        for (int i = len / 2; i >= 0; i--) {
            maxHeapify(nums, i, len);
        }
        for (int i = len; i > 0; i--) {
            swap(nums, i, 0);
            maxHeapify(nums, 0, --len);
        }
    }
}
```

# 2058.找出临界点之间的最小和最大距离

```java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public int[] nodesBetweenCriticalPoints(ListNode head) {
        int minDistance = -1;
        int maxDistance = -1;
        int[] result = new int[2];
        List<Integer> indexList = new ArrayList<>();
        int index = 1;
        int prefix = head.val;
        ListNode node = head.next;
        while (node != null && node.next != null) {
            if ((node.val > prefix && node.val > node.next.val) || (node.val < prefix && node.val < node.next.val)) {
                indexList.add(index);
            }
            index++;
            prefix = node.val;
            node = node.next;
        }
        if (indexList.size() > 1) {
            maxDistance = indexList.get(indexList.size() - 1) - indexList.get(0);
            minDistance = indexList.get(1) - indexList.get(0);
            for (int i = 0; i < indexList.size() - 1; i++) {
                minDistance = Math.min(minDistance, indexList.get(i + 1) - indexList.get(i));
            }
        }
        result[0] = minDistance;
        result[1] = maxDistance;
        return result;
    }
}
```

# 2074.反转偶数长度组的节点

```java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public ListNode reverseEvenLengthGroups(ListNode head) {
        Stack<ListNode> stack = new Stack<>();
        ListNode newHead = new ListNode(0, head);
        ListNode node = newHead;
        int index = 0;
        while (node != null && node.next != null) {
            index++;
            int tempCount = index;
            ListNode preNode = node;
            while (tempCount > 0 && node != null && node.next != null) {
                node = node.next;
                stack.push(node);
                tempCount--;
            }
            ListNode lastNode = node.next;
            if (stack.size() % 2 == 0) {
                while (!stack.empty()) {
                    ListNode tempNode = stack.pop();
                    preNode.next = tempNode;
                    preNode = preNode.next;
                    preNode.next = lastNode;
                    node = preNode;
                }
            } else {
                stack.clear();
            }
        }
        return newHead.next;
    }
}
```

```java
            sum += rowCosts[i];
        }
        if (startPos[0] != homePos[0]) {
            sum += rowCosts[homePos[0]];
        }
        for (int i = Math.min(startPos[1], homePos[1]) + 1; i < Math.max(startPos[1], homePos[1]); i++) {
            sum += colCosts[i];
        }
        if (startPos[1] != homePos[1]) {
            sum += colCosts[homePos[1]];
        }
        return sum;
    }
```

```java
class Solution {
    public int[] getAverages(int[] nums, int k) {
        int[] avg = new int[nums.length];
        Arrays.fill(avg, -1);
        int len = k * 2 + 1;
        long sum = 0;
        for (int i = 0; i < nums.length; i++) {
            sum += nums[i];
            if (i < len - 1) {
                continue;
            }
            avg[i - k] = (int) (sum / len);
            sum -= nums[i - len + 1];
        }
        return avg;
    }
}
```

```java
        int min = Integer.MAX_VALUE;
        int maxIndex = -1;
        int minIndex = -1;
        for (int i = 0; i < nums.length; i++) {
            if (nums[i] > max) {
                max = nums[i];
                maxIndex = i;
            }
            if (nums[i] < min) {
                min = nums[i];
                minIndex = i;
            }
        }
        int leftIndex = Math.min(maxIndex, minIndex);
        int rightIndex = Math.max(maxIndex, minIndex);

    }
}
```
