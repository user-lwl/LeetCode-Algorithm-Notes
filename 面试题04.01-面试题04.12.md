# 面试题 04.02.最小高度树

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    int[] num;

    public TreeNode sortedArrayToBST(int[] nums) {
        num = nums;
        return buildTree(0, num.length - 1);
    }

    public TreeNode buildTree(int left, int right) {
        if (left > right) {
            return null;
        }
        int index = (left + right) / 2;
        TreeNode node = new TreeNode(num[index]);
        node.left = buildTree(left, index - 1);
        node.right = buildTree(index + 1, right);
        return node;
    }
}
```

# 面试题 04.03.特定深度节点链表

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    ListNode[] result;
    List<List<Integer>> tempList = new ArrayList<>();
    public ListNode[] listOfDepth(TreeNode tree) {
        doIt(tree, 0);
        result = new ListNode[tempList.size()];
        for (int i = 0; i < tempList.size(); i++) {
            List<Integer> temp = tempList.get(i);
            ListNode tempNode = new ListNode(temp.get(0));
            result[i] = tempNode;
            for (int j = 1; j < temp.size(); j++) {
                int val = temp.get(j);
                ListNode tmpNode = new ListNode(val);
                tempNode.next = tmpNode;
                tempNode = tmpNode;
            }
        }
        return result;
    }
    public void doIt(TreeNode node, int level) {
        if (node == null) {
            return;
        }
        while (tempList.size() <= level) {
            tempList.add(new ArrayList<>());
        }
        List<Integer> temp = tempList.get(level);
        temp.add(node.val);
        doIt(node.left, level + 1);
        doIt(node.right, level + 1);
    }
}
```

# 面试题 04.04.检查平衡性

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    public boolean isBalanced(TreeNode root) {
        if (root == null) {
            return true;
        }
        return Math.abs(height(root.left) - height(root.right)) <= 1 && isBalanced(root.left) && isBalanced(root.right);
    }
    public int height(TreeNode node) {
        if (node == null) {
            return 0;
        }
        return Math.max(height(node.left), height(node.right)) + 1;
    }
}
```

# 面试题 04.05. 合法二叉搜索树

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    public boolean isValidBST(TreeNode root) {
        return isT(root, Long.MAX_VALUE, Long.MIN_VALUE);
    }
    public boolean isT(TreeNode node, long max, long min) {
        if (node == null) {
            return true;
        }
        if (node.val >= max || node.val <= min) {
            return false;
        }
        return isT(node.left, node.val, min) && isT(node.right, max, node.val);
    }
}
```



# 面试题 04.08.首个共同祖先

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    TreeNode result = null;
    TreeNode globalP;
    TreeNode globalQ;
    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        globalP = p;
        globalQ = q;
        findParent(root);
        return result;
    }
    public boolean[] findParent(TreeNode node) {
        if (node == null) {
            return new boolean[]{false, false};
        }
        boolean[] left = findParent(node.left);
        boolean[] right = findParent(node.right);
        boolean[] thisFlag = new boolean[]{left[0] | right[0], left[1] | right[1]};
        if (node == globalP) {
            thisFlag[0] = true;
        }
        if (node == globalQ) {
            thisFlag[1] = true;
        }
        if (thisFlag[0] && thisFlag[1] && result == null) {
            result = node;
        }
        return thisFlag;
    }
}
```
