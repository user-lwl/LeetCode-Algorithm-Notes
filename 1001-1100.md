```java
        for (int i = 0; i < words.length; i++) {

                char resultChar = (char) ('a' + i);
                resultList.add(String.valueOf(resultChar));
                min--;
            }
        }
        return resultList;
    }
}
```

```java
        int len = nums.length;
```

# 1019.链表中的下一个更大节点

```java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    public int[] nextLargerNodes(ListNode head) {
        ListNode node = head;
        Stack<ListNode> stack = new Stack<>();
        while (node != null) {
            stack.push(node);
            node = node.next;
        }
        int size = stack.size();
        int[] nodeValList = new int[size];
        int[] result = new int[size];
        int index = size - 2;
        result[size - 1] = 0;
        node = stack.pop();
        int max = node.val;
        nodeValList[size - 1] = node.val;
        while (!stack.empty()) {
            ListNode tempNode = stack.pop();
            nodeValList[index] = tempNode.val;
            if (tempNode.val >= max) {
                result[index--] = 0;
                max = tempNode.val;
            } else {
                for (int i = index; i < size; i++) {
                    if (nodeValList[i] > tempNode.val) {
                        result[index--] = nodeValList[i];
                        break;
                    }
                }
            }
        }
        return result;
    }
}
```

```java
class Solution {
```

```java
        if (node == null) {
            return;

    }
```

# 1079.活字印刷

```java
class Solution {

    List<List<Character>> resultList = new ArrayList<>();
    List<Character> tempList = new ArrayList<>();
    char[] tempChar;

    public int numTilePossibilities(String tiles) {
        tempChar = tiles.toCharArray();
        doIt(0);
        return resultList.size();
    }
    public void doIt(int index) {
        if (tempList.size() > 0) {
            resultList.add(new ArrayList<>(tempList));
        }
        if (index >= tempChar.length) {
            return;
        }
        Set<Character> tempSet = new HashSet<>();
        for (int i = index; i < tempChar.length; i++) {
            if (tempSet.contains(tempChar[i])) {
                continue;
            }
            tempSet.add(tempChar[i]);
            tempList.add(tempChar[i]);
            swap(index, i);
            doIt(index + 1);
            tempList.remove(tempList.size() - 1);
            swap(index, i);
        }
    }
    public void swap(int indexA, int indexB) {
        char temp = tempChar[indexA];
        tempChar[indexA] = tempChar[indexB];
        tempChar[indexB] = temp;
    }
}
```
