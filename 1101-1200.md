        int index = 1;
        while (candies > 0) {
            for (int i = 0; i < result.length; i++) {

                result[i] += temp;
                candies -= temp;
                index++;

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    TreeNode resut = null;
    int maxDep = 0;

    public TreeNode lcaDeepestLeaves(TreeNode root) {
        doHelper(root, 1);
        return resut;
    }
    public int doHelper(TreeNode node, int depth) {
        if (node == null) {
            return depth - 1;
        }
        int left = doHelper(node.left, depth + 1);
        int right = doHelper(node.right, depth + 1);
        if (left == right) {
            if (left >= maxDep) {
                maxDep = left;
                resut = node;
            }
        }
        return Math.max(left, right);
    }
}

        for (int i = 0; i < arr.length - 1; i++) {
            List<Integer> tempArray = new ArrayList<>();
            tempArray.add(arr[i]);
            tempArray.add(arr[i + 1]);
            int temp = arr[i + 1] - arr[i];
            List<List<Integer>> arrArray = tempMap.getOrDefault(temp, new ArrayList<>());
            arrArray.add(tempArray);
            tempMap.put(temp, arrArray);
        }
        int min = Integer.MAX_VALUE;
        for (Integer cha : tempMap.keySet()) {
            min = Math.min(cha, min);
        }
        return tempMap.get(min);
    }
