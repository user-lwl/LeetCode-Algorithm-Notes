```java
class Solution {
    public int divide(int a, int b) {
        int flag = 1;
        if (a > 0) {
            flag *= -1;
            a *= -1;
        }
        if (b > 0) {
            flag *= -1;
            b *= -1;
        }
        int result = 0;
        while (a <= b) {
            int tempB = b;
            int cnt = 1;
            while (a <= tempB << 1 && tempB >= Integer.MIN_VALUE >> 1) {
                tempB = tempB << 1;
                cnt = cnt << 1;
            }
            result += cnt;
            a -= tempB;
        }
        if (flag == 1 && result == Integer.MIN_VALUE) {
            return Integer.MAX_VALUE;
        }
        return result * flag;
    }
}
```

```java
class Solution {
    public String addBinary(String a, String b) {
        int[] aList = new int[Math.max(a.length(), b.length()) + 1];
        int[] bList = new int[Math.max(a.length(), b.length()) + 1];
        int indexA = a.length() - 1;
        int indexB = b.length() - 1;
        int isNext = 0;
        StringBuilder sb = new StringBuilder();
        for (int i = aList.length - 1; i >= 0; i--) {
            if (indexA >= 0) {
                aList[i] = a.charAt(indexA--) - '0';
            }
            if (indexB >= 0) {
                bList[i] = b.charAt(indexB--) - '0';
            }
            sb.append((aList[i] + bList[i] + isNext) % 2);
            isNext = (aList[i] + bList[i] + isNext) / 2;
        }
        if (sb.charAt(sb.length() - 1) == '0') {
            sb.deleteCharAt(sb.length() - 1);
        }
        return sb.reverse().toString();
    }
}
```

# LCR 005.最大单词长度乘积

```java
class Solution {
    public int maxProduct(String[] words) {
        int[] binary = new int[words.length];
        for (int i = 0; i < words.length; i++) {
            for (int j = 0; j < words[i].length(); j++) {
                binary[i] |= (1 << (words[i].charAt(j) - 'a'));
            }
        }
        int result = 0;
        for (int i = 0; i < binary.length; i++) {
            for (int j = i + 1; j < binary.length; j++) {
                if ((binary[i] & binary[j]) == 0) {
                    result = Math.max(result, words[i].length() * words[j].length());
                }
            }
        }
        return result;
    }
}
```

# LCR 031. LRU缓存

```java
class LRUCache {

    class Node {
        public int key;
        public int val;
        public Node prev;
        public Node next;
        Node() { }
        Node(int k, int v) {
            key = k;
            val = v;
        }
    }

    int size;
    int capacity;
    Node head;
    Node tail;
    Map<Integer, Node> map = new HashMap<>();

    public LRUCache(int capacity) {
        size = 0;
        this.capacity = capacity;
        head = new Node();
        tail = new Node();
        head.next = tail;
        tail.prev = head;
    }
    
    public int get(int key) {
        Node node = map.get(key);
        if (node == null) {
            return -1;
        }
        moveToHead(node);
        return node.val;
    }
    
    public void put(int key, int value) {
        Node node = map.get(key);
        if (node == null) {
            node = new Node(key, value);
            map.put(key, node);
            addToHead(node);
            size++;
            if (size > capacity) {
                Node remove = removeTail();
                map.remove(remove.key);
                size--;
            }
        } else {
            node.val = value;
            moveToHead(node);
        }
    }

    public void addToHead(Node node) {
        node.prev = head;
        node.next = head.next;
        head.next.prev = node;
        head.next = node;
    }

    public void moveToHead(Node node) {
        removeNode(node);
        addToHead(node);
    }

    public void removeNode(Node node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
    }

    public Node removeTail() {
        Node result = tail.prev;
        removeNode(result);
        return result;
    }
}
```

```java
 *     int val;
 *     ListNode next;

 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }

class Solution {
    public ListNode sortList(ListNode head) {
        if (head == null || head.next == null) {
            return head;
        }
        ListNode slow = head;
        ListNode fast = head;
        ListNode pre = head;
        while (fast != null) {
            pre = slow;
            slow = slow.next;
            fast = fast.next;
            if (fast != null) {
                fast = fast.next;
            }
        }
        ListNode mid = pre.next;
        pre.next = null;
        return mergeSort(sortList(head), sortList(mid));
    }

        ListNode result = new ListNode();
        ListNode temp = result;
        while (node1 != null && node2 != null) {
            if (node1.val < node2.val) {
                temp.next = node1;
                node1 = node1.next;
            } else {
                temp.next = node2;
                node2 = node2.next;
            }
            temp = temp.next;
        }
        if (node1 != null) {
            temp.next = node1;
        }
        if (node2 != null) {
            temp.next = node2;
        }
        return result.next;
    }
}
```
