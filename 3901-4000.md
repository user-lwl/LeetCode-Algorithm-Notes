# 3903.最小稳定下标Ⅰ

```java
class Solution {
    public int firstStableIndex(int[] nums, int k) {
        int[] max = new int[nums.length];
        int[] min = new int[nums.length];
        max[0] = nums[0];
        min[min.length - 1] = nums[nums.length - 1];
        for (int i = 1; i < nums.length; i++) {
            max[i] = Math.max(max[i - 1], nums[i]);
        }
        for (int i = nums.length - 2; i >= 0; i--) {
            min[i] = Math.min(min[i + 1], nums[i]);
        }
        for (int i = 0; i < nums.length; i++) {
            if (max[i] - min[i] <= k) {
                return i;
            }
        }
        return -1;
    }
}
```

# 3905.多源图像渲染

```java
class Solution {
    int[][] result;
    int[][] visited;
    Queue<int[]> queue = new LinkedList<>();
    public int[][] colorGrid(int n, int m, int[][] sources) {
        result = new int[n][m];
        visited = new int[n][m];
        for (int i = 0; i < visited.length; i++) {
            Arrays.fill(visited[i], Integer.MAX_VALUE);
        }
        for (int i = 0; i < sources.length; i++) {
            // x,y,color,time
            queue.add(new int[]{sources[i][0], sources[i][1], sources[i][2], 0});
            result[sources[i][0]][sources[i][1]] = sources[i][2];
            visited[sources[i][0]][sources[i][1]] = 0;
        }
        while (!queue.isEmpty()) {
            int[] temp = queue.poll();
            int x = temp[0];
            int y = temp[1];
            int color = temp[2];
            int time = temp[3];
            if (x > 0) {
                doIt(x - 1, y, color, time + 1);
            }
            if (x < result.length - 1) {
                doIt(x + 1, y, color, time + 1);
            }
            if (y > 0) {
                doIt(x, y - 1, color, time + 1);
            }
            if (y < result[0].length - 1) {
                doIt(x, y + 1, color, time + 1);
            }
        }
        return result;
    }
    public void doIt(int x, int y, int color, int time) {
        if (time < visited[x][y]) {
            result[x][y] = color;
            visited[x][y] = time;
            queue.add(new int[]{x, y, color, time + 1});
        } else if (time == visited[x][y] && result[x][y] < color) {
            result[x][y] = color;
            queue.add(new int[]{x, y, color, time + 1});
        }
    }
}
```

# 3908.有效数字整数

```java
class Solution {
    public boolean validDigit(int n, int x) {
        int t = n;
        while (t > 9) {
            t /= 10;
        }
        if (t == x) {
            return false;
        }
        while (n > 9) {
            int temp = n % 10;
            if (temp == x) {
                return true;
            }
            n /= 10;
        }
        return false;
    }
}
```

# 3909.比较双调部分的和

```java
class Solution {
    public int compareBitonicSums(int[] nums) {
        int index = 0;
        long left = 0;
        long right = 0;
        for (;index < nums.length; index++) {
            left += nums[index];
            if (nums[index] > nums[index + 1]) {
                break;
            }
        }
        for (;index < nums.length; index++) {
            right += nums[index];
        }
        return left == right ? -1 : (left > right ? 0 : 1);
    }
}
```

# 3910.统计节点和为偶数的连通子图

```java
class Solution {
    List<List<Integer>> tempList = new ArrayList<>();
    int[] global;
    List<Integer> temp = new ArrayList<>();
    int result = 0;
    public int evenSumSubgraphs(int[] nums, int[][] edges) {
        global = nums;
        for (int i = 0; i < nums.length; i++) {
            tempList.add(new ArrayList<>());
        }
        for (int i = 0; i < edges.length; i++) {
            List<Integer> list1 = tempList.get(edges[i][0]);
            List<Integer> list2 = tempList.get(edges[i][1]);
            list1.add(edges[i][1]);
            list2.add(edges[i][0]);
        }
        doIt(0, 0, 0);
        return result;
    }
    public void doIt(int index, int sum, int bg) {
        if (temp.size() != 0 && sum % 2 == 0 && bg == 1) {
            if (isConnected()) {
                result++;
            }
        }
        if (index >= tempList.size()) {
            return;
        }
        doIt(index + 1, sum, 0);
        temp.add(index);
        doIt(index + 1, sum + global[index], 1);
        temp.remove(temp.size() - 1);
    }
    public boolean isConnected() {
        List<Integer> thisTemp = temp;
        if (thisTemp.size() == 1) {
            return true;
        }
        Set<Integer> tempResSet = new HashSet<>();
        Queue<Integer> que = new LinkedList<>();
        tempResSet.add(thisTemp.get(0));
        que.add(thisTemp.get(0));
        while (!que.isEmpty()) {
            int ttt = que.poll();
            List<Integer> ttl = tempList.get(ttt);
            for (Integer key : ttl) {
                if (thisTemp.contains(key) && tempResSet.add(key)) {
                    que.add(key);
                }
            }
        }
        return tempResSet.size() == thisTemp.size();
    }
}
```

# 3912.数组中的有效元素

```java
class Solution {
    public List<Integer> findValidElements(int[] nums) {
        int[] prefix = new int[nums.length];
        int[] suffix = new int[nums.length];
        prefix[0] = nums[0];
        suffix[nums.length - 1] = nums[nums.length - 1];
        for (int i = 1; i < nums.length; i++) {
            prefix[i] = Math.max(prefix[i - 1], nums[i]);
        }
        for (int i = nums.length - 2; i >= 0; i--) {
            suffix[i] = Math.max(suffix[i + 1], nums[i]);
        }
        List<Integer> resultList = new ArrayList<>();
        resultList.add(nums[0]);
        for (int i = 1; i < nums.length - 1; i++) {
            if (nums[i] > prefix[i - 1] || nums[i] > suffix[i + 1]) {
                resultList.add(nums[i]);
            }
        }
        if (nums.length > 1) {
            resultList.add(nums[nums.length - 1]);
        }
        return resultList;
    }
}
```

# 3913.按频率对元音排序

```java
class Solution {
    public String sortVowels(String s) {
        int[] cntArray = new int[26];
        int[] firstIndex = new int[26];
        Arrays.fill(firstIndex, -1);
        List<Integer> indexList = new ArrayList<>();
        for (int i = 0; i < s.length(); i++) {
            char temp = s.charAt(i);
            if (temp != 'a' && temp != 'e' && temp != 'i' && temp != 'o' && temp != 'u') {
                continue;
            }
            cntArray[temp - 'a']++;
            indexList.add(i);
            if (firstIndex[temp - 'a'] == -1) {
                firstIndex[temp - 'a'] = i;
            }
        }
        int[][] sortArray = new int[5][3];
        sortArray[0][0] = 'a';
        sortArray[0][1] = cntArray['a' - 'a'];
        sortArray[0][2] = firstIndex['a' - 'a'];
        sortArray[1][0] = 'e';
        sortArray[1][1] = cntArray['e' - 'a'];
        sortArray[1][2] = firstIndex['e' - 'a'];
        sortArray[2][0] = 'i';
        sortArray[2][1] = cntArray['i' - 'a'];
        sortArray[2][2] = firstIndex['i' - 'a'];
        sortArray[3][0] = 'o';
        sortArray[3][1] = cntArray['o' - 'a'];
        sortArray[3][2] = firstIndex['o' - 'a'];
        sortArray[4][0] = 'u';
        sortArray[4][1] = cntArray['u' - 'a'];
        sortArray[4][2] = firstIndex['u' - 'a'];
        char[] tempChar = s.toCharArray();
        Arrays.sort(sortArray, (a, b) -> {
            if (a[1] == b[1]) {
                return Integer.compare(a[2], b[2]);
            }
            return Integer.compare(b[1], a[1]);
        });
        int index = 0;
        int sortArrayIndex = 0;
        for (;index < indexList.size(); index++) {
            while (sortArray[sortArrayIndex][1] == 0) {
                sortArrayIndex++;
            }
            tempChar[indexList.get(index)] = (char) sortArray[sortArrayIndex][0];
            sortArray[sortArrayIndex][1]--;
        }
        return new String(tempChar);
    }
}
```

# 3914.使数组非递减需要的最小累计值

```java
class Solution {
    public long minOperations(int[] nums) {
        long result = 0;
        for (int i = 0; i < nums.length - 1; i++) {
            if (nums[i + 1] < nums[i]) {
                result += nums[i] - nums[i + 1];
            }
        }
        return result;
    }
}
```
