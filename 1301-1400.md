# 1302.层数最深叶子节点的和

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    int maxLevel = 0;
    int sum = 0;

    public int deepestLeavesSum(TreeNode root) {
        doHelper(root, 1);
        return sum;
    }
    public void doHelper(TreeNode node, int level) {
        if (node == null) {
            return;
        }
        if (level > maxLevel) {
            maxLevel = level;
            sum = 0;
        }
        if (node.left == null && node.right == null && maxLevel == level) {
            sum += node.val;
        }
        doHelper(node.left, level + 1);
        doHelper(node.right, level + 1);
    }
}
```

# 1305.两棵二叉搜索树中的所有元素

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    List<Integer> result = new ArrayList<>();
    List<Integer> listA = new ArrayList<>();
    List<Integer> listB = new ArrayList<>();

    public List<Integer> getAllElements(TreeNode root1, TreeNode root2) {
        doHelper(root1, listA);
        doHelper(root2, listB);
        int indexA = 0;
        int indexB = 0;
        while (indexA < listA.size() && indexB < listB.size()) {
            if (listA.get(indexA) < listB.get(indexB)) {
                result.add(listA.get(indexA++));
            } else {
                result.add(listB.get(indexB++));
            }
        }
        while (indexA < listA.size()) {
            result.add(listA.get(indexA++));
        }
        while (indexB < listB.size()) {
            result.add(listB.get(indexB++));
        }
        return result;
    }

    public void doHelper(TreeNode node, List<Integer> list) {
        if (node == null) {
            return;
        }
        doHelper(node.left, list);
        list.add(node.val);
        doHelper(node.right, list);
    }
}
```

# 1310.子数组异或查询

```java
class Solution {
    public int[] xorQueries(int[] arr, int[][] queries) {
        int len = arr.length;
        int[] prefix = new int[len];
        int[] result = new int[queries.length];
        prefix[0] = arr[0];
        for (int i = 1; i < len; i++) {
            prefix[i] = prefix[i - 1] ^ arr[i];
        }
        for (int i = 0; i < queries.length; i++) {
            int left = queries[i][0];
            int right = queries[i][1];
            if (left == 0) {
                result[i] = prefix[right];
            } else {
                result[i] = prefix[right] ^ prefix[left - 1];
            }
        }
        return result;
    }
}
```

```java
        int m = mat.length;
        int n = mat[0].length;
        int[][] result = new int[m][n];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                result[i][j] = fix(mat, i, j, k);
            }
        }
        return result;
    }

    public int fix(int[][] mat, int i, int j, int k) {
        int res = 0;
        for (int r = i - k; r <= i+k ;r++) {
            for (int c = j - k; c <= j+k; c++) {
                if (0 <= r && r < mat.length && 0 <= c && c < mat[0].length) {
                    res += mat[r][c];
                }
            }
        }
        return res;
    }
}
```

```java
 *     int val;
 *     TreeNode left;

 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }

 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }

class Solution {

    int result = 0;

    public int sumEvenGrandparent(TreeNode root) {
        if (root.left != null) {
            doHelper(root.left.left, root.left.val, root.val);
            doHelper(root.left.right, root.left.val, root.val);
        }
        if (root.right != null) {
            doHelper(root.right.left, root.right.val, root.val);
            doHelper(root.right.right, root.right.val, root.val);
        }
        return result;
    }
    public void doHelper(TreeNode node, int fater, int grandFather) {
        if (node == null) {
            return;
        }
        if (grandFather % 2 == 0) {
            result += node.val;
        }
        doHelper(node.left, node.val, fater);
        doHelper(node.right, node.val, fater);
    }
}
```

```java
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }

 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }

class Solution {

    int totalTarget;

    public TreeNode removeLeafNodes(TreeNode root, int target) {
        totalTarget = target;
        doHelper(root.left, root, 0);
        doHelper(root.right, root, 1);
        if (root.val == totalTarget && root.left == null && root.right == null) {
            root = null;
        }
        return root;
    }
    // fx 0-left 1-right
    public void doHelper(TreeNode node, TreeNode father, int fx) {
        if (node == null) {
            return;
        }
        doHelper(node.left, node, 0);
        doHelper(node.right, node, 1);
        if (node.left == null && node.right == null && node.val == totalTarget) {
            if (fx == 0) {
                father.left = null;
            } else {
                father.right = null;
            }
        }
    }
}
```

```java
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {

    public boolean isSubPath(ListNode head, TreeNode root) {
        if (root == null) {
            return false;
        }
        return isT(head, root) || isSubPath(head, root.left) || isSubPath(head, root.right);
    }

    public boolean isT(ListNode head, TreeNode root) {
        if (root == null) {
            return false;
        }
        if (root.val == head.val) {
            if (head.next == null) {
                return true;
            } else {
                return isT(head.next, root.left) || isT(head.next, root.right);
            }
        }
        return false;
    }
}
```

# 1371.每个元音包含偶数次的最长子字符串

```java
class Solution {
    public int findTheLongestSubstring(String s) {
        int[] prefix = new int[s.length() + 1];
        int result = 0;
        Map<Integer, Integer> tempMap = new HashMap<>();
        Set<Character> yuanSet = Set.of('a', 'e', 'i', 'o', 'u');
        for (int i = 1; i < prefix.length; i++) {
            prefix[i] = prefix[i - 1];
            char tempChar = s.charAt(i - 1);
            if (yuanSet.contains(tempChar)) {
                prefix[i] = prefix[i] ^ (tempChar - 'a' + 1);
            }
            Integer ttmp = tempMap.getOrDefault(prefix[i], null);
            if (prefix[i] == 0) {
                result = Math.max(result, i);
            } else if (ttmp != null) {
                result = Math.max(result, i - ttmp);
            }
            if (ttmp == null) {
                tempMap.put(prefix[i], i);
            }
        }
        return result;
    }
}
```

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */

class Solution {
    public final TreeNode getTargetCopy(final TreeNode original, final TreeNode cloned, final TreeNode target) {
        return getCopyTarget(cloned, target);
    }

    public TreeNode getCopyTarget(TreeNode cloned, TreeNode target) {
        if (cloned == null) {
            return null;
        }
        if (cloned.val == target.val) {
            return cloned;
        }
        TreeNode leftNode = getCopyTarget(cloned.left, target);
        TreeNode rightNode = getCopyTarget(cloned.right, target);
        if (leftNode == null) {
            return rightNode;
        }
        return leftNode;
    }
}
```
