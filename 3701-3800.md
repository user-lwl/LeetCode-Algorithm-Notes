```java
class Solution {
    public int longestSubsequence(int[] nums) {
        Arrays.sort(nums);
        if (nums[nums.length - 1] == 0) {
            return 0;
        }
        int result = nums[0];
        int len = nums.length;
        int index = nums.length - 1;
        for (int i = 1; i < nums.length; i++) {
            result ^= nums[i];
        }
        while (result == 0) {
            result ^= nums[index--];
            len--;
        }
        return len;
    }
```

```java
    public long maxAlternatingSum(int[] nums) {
        long[] temp = new long[nums.length];
        for (int i = 0; i < nums.length; i++) {
            temp[i] = (long) nums[i] * nums[i];
        }
        Arrays.sort(temp);
        int left = 0;
        int right = temp.length - 1;
        long result = 0;
        while (left < right) {
            result += temp[right--];
            result -= temp[left++];

        }
        return result;
    }
```

```java
            tempNumber[i][0] = nums[i];
```

# 3776.使循环数组余额非负的最少移动次数

```java
class Solution {
    public long minMoves(int[] balance) {
        int fuIndex = -1;
        for (int i = 0; i < balance.length; i++) {
            if (balance[i] < 0) {
                fuIndex = i;
                break;
            }
        }
        if (fuIndex == -1) {
            return 0;
        }
        long result = 0;
        for (int i = 1; i <= balance.length / 2; i++) {
            int leftIndex = (fuIndex - i + balance.length) % balance.length;
            int rightIndex = (fuIndex + i) % balance.length;
            int leftNeed = Math.min(balance[leftIndex], -balance[fuIndex]);
            result += (long) leftNeed * i;
            balance[fuIndex] += leftNeed;
            balance[leftIndex] -= leftNeed;
            int rightNeed = Math.min(balance[rightIndex], -balance[fuIndex]);
            result += (long) rightNeed * i;
            balance[fuIndex] += rightNeed;
            balance[rightIndex] -= rightNeed;
            if (balance[fuIndex] == 0) {
                break;
            }
        }
        if (balance[fuIndex] < 0) {
            return -1;
        }
        return result;
    }
}
```

```java
class Solution {
    public int findMaxVal(int n, int[][] restrictions, int[] diff) {
        int[] tempList = new int[n];
        int maxVal = 0;
        Set<Integer> specialIndex = new HashSet<>();
        for (int i = 0; i < restrictions.length; i++) {
            tempList[restrictions[i][0]] = restrictions[i][1];
            specialIndex.add(restrictions[i][0]);
        }
        for (int i = 1; i < tempList.length; i++) {
            if (specialIndex.contains(i)) {
                tempList[i] = Math.min(tempList[i - 1] + diff[i - 1], tempList[i]);
                continue;
            }
            tempList[i] = tempList[i - 1] + diff[i - 1];
        }
        maxVal = tempList[tempList.length - 1];
        for (int i = tempList.length - 2; i >= 0; i--) {
            if (specialIndex.contains(i)) {
                tempList[i] = Math.min(tempList[i + 1] + diff[i], tempList[i]);
            } else {
                int temp = tempList[i + 1] + diff[i];
                tempList[i] = Math.min(tempList[i], temp);
            }
            maxVal = Math.max(maxVal, tempList[i]);
        }
        return maxVal;
    }
}
```
