class Solution {
    public int longestMonotonicSubarray(int[] nums) {
        int result = 1;
        for (int i = 0; i < nums.length - 1; i++) {
            if (nums[i] > nums[i + 1]) {
                for (int j = i + 1; j < nums.length; j++) {
                    if (nums[j - 1] <= nums[j]) {
                        result = Math.max(result, j - i);
                        break;
                    }
                    if (j == nums.length - 1) {
                        result = Math.max(result, j - i + 1);
                        break;
                    }
                }
            } else if (nums[i] < nums[i + 1]) {
                for (int j = i + 1; j < nums.length; j++) {
                    if (nums[j - 1] >= nums[j]) {
                        result = Math.max(result, j - i);
                        break;
                    }
                    if (j == nums.length - 1) {
                        result = Math.max(result, j - i + 1);
                        break;
                    }
                }
            }
        }
        return result;
    }

        long result = 0;
        if (nums[midIndex] == k) {
            return 0;
        } else if (nums[midIndex] > k) {
            for (int i = midIndex; i >= 0; i--) {
                if (nums[i] <= k) {
                    break;
                }
                result += nums[i] - k;
            }
        } else {
            for (int i = midIndex; i < nums.length; i++) {
                if (nums[i] >= k) {
                    break;
                }
                result += k - nums[i];
            }
        }
        return result;
    }
