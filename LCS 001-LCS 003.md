# LCS 01.下载插件

```java
class Solution {
    public int leastMinutes(int n) {
        int size = 1;
        int minute = 0;
        while (size < n) {
            size *= 2;
            minute++;
        }
        return minute + 1;
    }
}
```

# LCS 02.完成一半的题目

```java
class Solution {
    public int halfQuestions(int[] questions) {
        Integer[] tempList = new Integer[1001];
        for (int i = 0; i < tempList.length; i++) {
            tempList[i] = 0;
        }
        int problem = questions.length;
        int person = problem / 2;
        for (int i = 0; i < questions.length; i++) {
            tempList[questions[i]]++;
        }
        Arrays.sort(tempList, Collections.reverseOrder());
        int cnt = 0;
        for (int i = 0; i < tempList.length; i++) {
            int problemNumber = tempList[i];
            if (problemNumber >= person) {
                return cnt + 1;
            }
            cnt++;
            person -= problemNumber;
        }
        return cnt;
    }
}
```

# LCS 03.主题空间

```java
class Solution {

    char[][] ttGrid;
    boolean[][] visited;

    public int largestArea(String[] grid) {
        int result = 0;
        ttGrid = new char[grid.length][grid[0].length()];
        visited = new boolean[grid.length][grid[0].length()];
        for (int i = 0; i < grid.length; i++) {
            ttGrid[i] = grid[i].toCharArray();
        }
        for (int i = 0; i < ttGrid.length; i++) {
            for (int j = 0; j < ttGrid[i].length; j++) {
                if (ttGrid[i][j] != '0' && !visited[i][j]) {
                    int tempRes = findRoom(i, j, ttGrid[i][j]);
                    if (tempRes != -1) {
                        result = Math.max(result, tempRes);
                    }
                }
            }
        }
        return result;
    }

    public int findRoom(int x, int y, char target) {
        if (ttGrid[x][y] == '0') {
            return -1;
        }
        if (ttGrid[x][y] != target || visited[x][y]) {
            return 0;
        }
        int result = 1;
        visited[x][y] = true;
        int left = 0;
        int right = 0;
        int top = 0;
        int bottom = 0;
        if (x > 0) {
            top = findRoom(x - 1, y, target);
        }
        if (y > 0) {
            left = findRoom(x, y - 1, target);
        }
        if (x < ttGrid.length - 1) {
            bottom = findRoom(x + 1, y, target);
        }
        if (y < ttGrid[0].length - 1) {
            right = findRoom(x, y + 1, target);
        }
        if (x == 0 || y == 0 || x == ttGrid.length - 1 || y == ttGrid[0].length - 1) {
            result = -1;
        }
        if (left == -1 || right == -1 || bottom == -1 || top == -1 || result == -1) {
            return -1;
        }
        return result + left + right + top + bottom;
    }
}
```
