```java
class Solution {

    int[] totalNums;
    
    public int minRemovals(int[] nums, int target) {
        int result = -1;
        totalNums = nums;
        Map<Integer, Integer> leftMap = new HashMap<>();
        Map<Integer, Integer> rightMap = new HashMap<>();
        // 左
        doHelper(0, 0, 0, nums.length / 2, leftMap);
        // 右
        doHelper(nums.length / 2 + 1, 0, 0, nums.length - 1, rightMap);
        for (Integer key : leftMap.keySet()) {
            int leftCnt = leftMap.get(key);
            int needNumber = target ^ key;
            if (rightMap.containsKey(needNumber)) {
                int rightCnt = rightMap.get(needNumber);
                result = Math.max(result, leftCnt + rightCnt);
            }
        }
        return result == -1 ? result : nums.length - result;
    }
    public void doHelper(int index, int xorSum, int keepCnt, int endIndex, Map<Integer, Integer> map) {
        if (index > endIndex){
            map.put(xorSum, Math.max(map.getOrDefault(xorSum, 0), keepCnt));
            return;
        }
        // 不用
        doHelper(index + 1, xorSum, keepCnt, endIndex, map);
        // 用
        doHelper(index + 1, xorSum ^ totalNums[index], keepCnt + 1, endIndex, map);
    }
}
```

```java
class Solution {

    int mod = (int) 1e9 + 7;
    
    public int countVisiblePeople(int n, int pos, int k) {
        // n-1 中选择 k
        // C(n, k) = n * ... * (n - k) + 1 / k!
        // (a / b) % mod = a * b^(-1) % mod
        k = Math.min(n - 1 - k, k);
        long result = doJie(n - k, n - 1, k);
        return (int) ((result * 2) % mod);
    }
    public long doJie(int start, int end, int k) {
        long result = 1;
        for (int i = start; i <= end; i++) {
            result *= i;
            result %= mod;
            long temp = (long) (k - i + start);
            long ivkTmp = ivk(temp);
            result *= ivkTmp;
            result %= mod;
        }
        return result;
    }
    // 乘法逆元代替除法
    public long ivk(long a) {
        return pow(a, mod - 2);
    }
    public long pow(long a, long b) {
        long result = 1;
        a %= mod;
        while (b > 0) {
            if ((b & 1) == 1) {
                result *= a;
                result %= mod;
            }
            a *= a;
            a %= mod;
            b >>= 1;
        }
        return result;
    }
```

```java
                }
            }
        }
        return false;
    }
```

# 3889.镜像频次距离

```java
class Solution {
    public int mirrorFrequency(String s) {
        int[] diag = new int[26];
        int[] nums = new int[10];
        for (int i = 0; i < s.length(); i++) {
            char temp = s.charAt(i);
            if (temp >= '0' && temp <= '9') {
                nums[temp - '0']++;
            } else {
                diag[temp - 'a']++;
            }
        }
        int result = 0;
        for (int i = 0; i <= 12; i++) {
            result += Math.abs(diag[i] - diag[25 - i]);
        }
        for (int i = 0; i <= 4; i++) {
            result += Math.abs(nums[i] - nums[9 - i]);
        }
        return result;
    }
}
```

# 3890.可由多种立方和构造的整数

```java
class Solution {
    public List<Integer> findGoodIntegers(int n) {
        List<Integer> resultList = new ArrayList<>();
        Set<Integer> tempSet = new HashSet<>();
        Set<Integer> haveSet = new HashSet<>();
        for (int i = 0; i * i * i <= n; i++) {
            for (int j = i; i * i * i + j * j * j <= n; j++) {
                int tmp = i * i * i + j * j * j;
                if (!tempSet.add(tmp)) {
                    if (haveSet.add(tmp)) {
                        resultList.add(tmp);
                    }
                }
            }
        }
        Collections.sort(resultList);
        return resultList;
    }
}
```

# 3891.最大化特殊下标数目的最少增加次数

```java
class Solution {
    public long minIncrease(int[] nums) {
        long result1 = 0;
        int n = nums.length;
        if (n <= 2) {
            return 0;
        }
        // 奇数
        if (n % 2 == 1) {
            for (int i = 1; i < n - 1; i += 2) {
                result1 += Math.max(Math.max(nums[i - 1], nums[i + 1]) - nums[i] + 1, 0);
            }
            return result1;
        }
        // 花费 - 偶数使用
        long[] cost = new long[n];
        long res2 = 0;
        long ji = 0;
        long ou = 0;
        for (int i = 1; i < n - 1; i++) {
            // 计算花费
            cost[i] = Math.max(Math.max(nums[i - 1], nums[i + 1]) - nums[i] + 1, 0);
            if (i % 2 == 0) {
                res2 += cost[i];
                ou += cost[i];
            } else {
                ji += cost[i];
            }
        }
        long res1 = 0;
        long minRes1 = Long.MAX_VALUE;
        for (int i = 1; i < n - 1; i+=2) {
            res1 += cost[i];
            ji -= cost[i];
            ou -= cost[i + 1];
            minRes1 = Math.min(minRes1, res1 + Math.min(ji, ou));
        }
        return Math.min(minRes1, res2);
    }
}
```
